from kafka import KafkaConsumer
import json

# Connect to Kafka broker
consumer = KafkaConsumer(
    'server_metrics',
    bootstrap_servers='localhost:9092',
    auto_offset_reset='earliest',
    enable_auto_commit=True,
    value_deserializer=lambda x: json.loads(x.decode('utf-8'))
)

print("Consumer started...")
print("Waiting for server metrics...\n")

# Continuously receive messages
for message in consumer:
    data = message.value

    server_id = data["server_id"]
    cpu = data["cpu_usage"]
    memory = data["memory_usage"]

    print("Received:")
    print(f"Server: {server_id}")
    print(f"CPU: {cpu}%")
    print(f"Memory: {memory}%")
    print()

    # Detect high CPU
    if cpu > 80:
        print(f"ALERT: High CPU detected on {server_id}")
        print()

# ------------------------------------------------

    (df["Memory"] > MEMORY_THRESHOLD) |

)

anomalies = df[df["Anomaly"]]

    )

from airflow import DAG
from airflow.operators.python import PythonOperator#from airflow.providers.standard.operators.python import # PythonOperator
from datetime import datetime


# Task 1: Collect Metrics
def collect_metrics(**context):
    metrics = {
        "cpu": 87,
        "memory": 65,
        "response_time": 420
    }

    print("Metrics collected:")
    print("CPU:", metrics["cpu"], "%")
    print("Memory:", metrics["memory"], "%")
    print("Response Time:", metrics["response_time"], "ms")

    # Store metrics for next task
    context["ti"].xcom_push(key="metrics", value=metrics)


# Task 2: Process Metrics
def process_metrics(**context):
    metrics = context["ti"].xcom_pull(
        task_ids="collect_metrics",
        key="metrics"
    )

    print("Processing metrics...")
    print("CPU:", metrics["cpu"], "%")
    print("Memory:", metrics["memory"], "%")
    print("Response Time:", metrics["response_time"], "ms")


# Task 3: Detect Anomaly
def detect_anomaly(**context):
    metrics = context["ti"].xcom_pull(
        task_ids="collect_metrics",
        key="metrics"
    )

    if metrics["cpu"] > 80:
        print("Anomaly detected: High CPU usage")
    else:
        print("No anomaly detected")


# Task 4: Generate Report
def generate_report():
    print("===== AIOps Report =====")
    print("Metrics collected successfully")
    print("Metrics processed successfully")
    print("Anomaly detection completed")
    print("========================")


# Create DAG
with DAG(
    dag_id="aiops_basic_workflow",
    start_date=datetime(2026, 9, 20),
    schedule=None,
    catchup=False
) as dag:

    collect_metrics_task = PythonOperator(
        task_id="collect_metrics",
        python_callable=collect_metrics
    )

    process_metrics_task = PythonOperator(
        task_id="process_metrics",
        python_callable=process_metrics
    )

    detect_anomaly_task = PythonOperator(
        task_id="detect_anomaly",
        python_callable=detect_anomaly
    )

    generate_report_task = PythonOperator(
        task_id="generate_report",
        python_callable=generate_report
    )

    # Dependencies
    collect_metrics_task >> process_metrics_task >> detect_anomaly_task >> generate_report_task
